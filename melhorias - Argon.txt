    return {

        get currentIndex() {

        },
        wrap(min, max, v) {
            const rangeSize = max - min;
            return ((((v - min) % rangeSize) + rangeSize) % rangeSize) + min;
        },
        getChipStyle(index) {
            const distance = index - this.currentIndex;
            const wrappedDistance = this.wrap(-(this.len / 2), this.len / 2, distance);
            return `transform: translateY(${wrappedDistance * this.itemHeight}px); opacity: ${1 - Math.abs(wrappedDistance) * 0.25}; height: ${this.itemHeight}px; width: max-content; transition: transform 0.7s cubic-bezier(0.175, 0.885, 0.32, 1.275), opacity 0.7s cubic-bezier(0.175, 0.885, 0.32, 1.275);`;
        },
        getCardStatus(index) {
            const diff = index - this.currentIndex;
            let normalizedDiff = diff;
            if (diff > this.len / 2) normalizedDiff -= this.len;
            if (diff < -this.len / 2) normalizedDiff += this.len;
            
            if (normalizedDiff === 0) return 'active';
            if (normalizedDiff === -1) return 'prev';
            if (normalizedDiff === 1) return 'next';
